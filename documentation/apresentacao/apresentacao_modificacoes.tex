\begin{frame}{1 · O traçado impresso}
  \mensagem{Impressoras reais não desenham uma linha perfeita}
  \centering
  \antes{0.31\textwidth}{linha perfeita e uniforme}{a_traco_antes}\hfill
  \depois{0.31\textwidth}{espessura varia ao longo da linha}{a_traco_espessura}\hfill
  \depois{0.31\textwidth}{com falhas de tinta}{a_traco_falhas}

  \vspace{0.35cm}
  \raggedright
  \begin{itemize}
    \item A espessura muda como a pressão de uma caneta
    \item A tinta falha em alguns trechos: às vezes some, às vezes só fica mais clara
  \end{itemize}
\end{frame}

\begin{frame}{1 · A organização da folha}
  \mensagem{Aparelhos diferentes organizam a folha de formas diferentes}
  \centering
  \antes{0.46\textwidth}{marca preta entre colunas}{a_layout_antes}\hfill
  \depois{0.46\textwidth}{espaço entre colunas, nome afastado}{a_layout_depois}

  \vspace{0.35cm}
  \raggedright
  \begin{itemize}
    \item O espaço entre as colunas de derivações varia de folha para folha
    \item O nome da derivação (aVR, V1\dots) fica mais perto ou mais longe do traçado
  \end{itemize}
\end{frame}

% ---------------------------------------------------------------------------------------
\begin{frame}{2 · O papel}
  \mensagem{Papel de verdade é manuseado: dobra, amassa e faz sombra}
  \centering
  \antes{0.46\textwidth}{papel perfeitamente plano}{b_amassado_antes}\hfill
  \depois{0.46\textwidth}{papel amassado}{b_amassado_depois}

  \vspace{0.35cm}
  \raggedright
  \begin{itemize}
    \item As dobras entortam a grade milimetrada e criam sombras
    \item As sombras respeitam a direção de onde vem a luz, como numa foto real
  \end{itemize}
\end{frame}

% ---------------------------------------------------------------------------------------
\begin{frame}{3 · O foco da câmera}
  \mensagem{Fotos de celular raramente têm foco perfeito}
  \centering
  \antes{0.46\textwidth}{foco perfeito, artificial}{c_blur_antes}\hfill
  \depois{0.46\textwidth}{foco levemente suave}{c_blur_depois}

  \vspace{0.35cm}
  \raggedright
  \begin{itemize}
    \item Um desfoque leve suaviza as bordas sem apagar o traçado
  \end{itemize}
\end{frame}

% ---------------------------------------------------------------------------------------
\begin{frame}{4 · A luz do ambiente}
  \mensagem{A luz de uma sala nunca é igual em toda a folha}
  \centering
  \antes{0.31\textwidth}{luz uniforme}{d_luz_antes}\hfill
  \depois{0.31\textwidth}{luz vindo de um lado}{d_luz_iluminacao}\hfill
  \depois{0.31\textwidth}{cantos mais escuros}{d_luz_vinheta}

  \vspace{0.35cm}
  \raggedright
  \begin{itemize}
    \item Uma janela ou lâmpada deixa um lado da folha mais claro que o outro
    \item A lente da câmera escurece levemente os cantos da foto
  \end{itemize}
\end{frame}

% ---------------------------------------------------------------------------------------
\begin{frame}{5 · O brilho da foto}
  \mensagem{Cada foto sai um pouco mais clara ou mais escura}
  \centering
  \antes{0.31\textwidth}{original}{d_exp_100}\hfill
  \depois{0.31\textwidth}{mais escura}{d_exp_050}\hfill
  \depois{0.31\textwidth}{mais clara}{d_exp_115}

  \vspace{0.35cm}
  \raggedright
  \begin{itemize}
    \item Limitamos o quanto a foto pode clarear: em excesso, a \textbf{grade milimetrada some
          antes do traçado}, e sem a grade não é possível medir o exame
  \end{itemize}
\end{frame}

% ---------------------------------------------------------------------------------------
\begin{frame}{5 · A cor da luz}
  \mensagem{A cor da luz agora varia como na vida real}
  \centering
  \antes{0.23\textwidth}{gerador antigo: laranja forte}{d_temp_2778}\hfill
  \antes{0.23\textwidth}{gerador antigo: azul forte}{d_temp_19560}\hfill
  \depois{0.23\textwidth}{agora: luz mais fria}{d_wb_frio}\hfill
  \depois{0.23\textwidth}{agora: luz mais quente}{d_wb_quente}

  \vspace{0.35cm}
  \raggedright
  \begin{itemize}
    \item O gerador antigo sorteava entre laranja forte e azul forte, sem meio-termo
    \item Agora a cor segue a luz natural, de levemente fria a levemente quente
  \end{itemize}
\end{frame}

% ---------------------------------------------------------------------------------------
\begin{frame}{6 · A nitidez}
  \mensagem{Linhas finas precisam aparecer como numa câmera, sem serrilhado}
  \centering
  \antes{0.38\textwidth}{baixa resolução: grade falhada}{e_ss1}\hspace{0.6cm}
  \depois{0.38\textwidth}{mais detalhe, depois reduzida}{e_ss2}

  \vspace{0.35cm}
  \raggedright
  \begin{itemize}
    \item A imagem é desenhada com mais detalhe e depois reduzida, como faz o sensor de uma câmera
    \item As linhas finas da grade ficam contínuas, sem falhas nem serrilhado
  \end{itemize}
\end{frame}

% ---------------------------------------------------------------------------------------
\begin{frame}{6 · A granulação da câmera}
  \mensagem{Toda foto tem uma leve granulação, mas sutil}
  \centering
  \antes{0.34\textwidth}{gerador antigo: chuvisco forte}{e_ruido_imgaug}\hspace{0.8cm}
  \depois{0.34\textwidth}{agora: granulação sutil}{e_ruido_sensor}

  \vspace{0.3cm}
  \raggedright
  \begin{itemize}
    \item O chuvisco antigo era mais forte que o da foto mais ruidosa entre as 8.793 fotos reais
    \item Agora a granulação fica dentro do que se vê nas fotos reais
  \end{itemize}
\end{frame}

\begin{frame}{7 · A ordem das derivações}
  \mensagem{As derivações saem na mesma ordem do aparelho de ECG}

  \vspace{0.35cm}

    \item Assim o programa aprende a ordem que ele vai encontrar nos exames do INC
  \end{itemize}
